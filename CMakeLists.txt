set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(RLIOS_WARNINGS_AS_ERRORS "Treat warnings as errors in core" ON)

# Exactness rules (CLAUDE.md section 5): no FMA contraction, no fast-math, ever.
function(rlios_core_flags tgt)
  target_include_directories(${tgt} PUBLIC ${CMAKE_SOURCE_DIR}/core)
  target_compile_options(${tgt} PRIVATE -ffp-contract=off -fno-fast-math -fno-finite-math-only
    -Wall -Wextra -Wshadow -Wdouble-promotion)
  if(RLIOS_WARNINGS_AS_ERRORS)
    target_compile_options(${tgt} PRIVATE -Werror)
  endif()
endfunction()

  enable_language(OBJCXX)
  add_library(rlios_render STATIC core/render/renderer.mm)

# Exactness harness (CLAUDE.md section 9). Reference recordings of the real game are never
# committed: point RLIOS_REFERENCE_DIR at them. Tests report "skipped" where they are absent (CI).
add_executable(ball_harness tools/harness/ball_harness.cpp)
rlios_core_flags(ball_harness)
target_link_libraries(ball_harness PRIVATE rlios_sim rlios_io)
if(DEFINED ENV{RLIOS_REFERENCE_DIR})
  set(RLIOS_REFERENCE_DIR "$ENV{RLIOS_REFERENCE_DIR}" CACHE PATH "Reference recordings")
else()
  set(RLIOS_REFERENCE_DIR "${CMAKE_SOURCE_DIR}/../work/reference" CACHE PATH "Reference recordings")
endif()
foreach(scn ball_drop_500uu ball_roll_a ball_wall_px ball_ceiling)
  add_test(NAME exact_${scn} COMMAND ball_harness ${RLIOS_REFERENCE_DIR}/${scn})
  set_tests_properties(exact_${scn} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
